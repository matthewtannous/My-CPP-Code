void userChoice(Tree &tree, int choice)
{
    ElementType element;
    switch (choice)
    {
    case 1:
        cout << "Enter element to insert: ";
        cin >> element;
        if (tree.insert(element))
        {
            cout << "Insertion successful\n";
        }
        else
        {
            cout << "Insertion failed\n";
        }
        break;
    case 2:
        cout << "In order: ";
        tree.inOrder(cout);
        break;
    case 3:
        cout << "Pre order: ";
        tree.preOrder(cout);
        break;
    case 4:
        cout << "Post order: ";
        tree.postOrder(cout);
        break;
    case 5:
        cout << "Enter element to check: ";
        cin >> element;
        if (tree.search(element))
        {
            cout << "Element " << element << " is in the tree\n";
        }
        else
        {
            cout << "Element " << element << " is not in the tree\n";
        }
        break;
    case 6:
        cout << "Number of elements: " << tree.countNodes();
        break;
    case 7:
        cout << "Sum of nodes: " << tree.getSum();
        break;
    case 8:
        cout << "Enter element to delete: ";
        cin >> element;
        if (tree.remove(element))
        {
            cout << "Deletion successful\n";
        }
        else
        {
            cout << "Deletion failed\n";
        }
        break;
    case 0:
        cout << "BYE\n";
        return;
    default:
        cout << "Invalid choice.\n";
    }
}
